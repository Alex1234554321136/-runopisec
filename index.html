// ===== ЧАСТЬ 4: ДОПОЛНИТЕЛЬНЫЕ ФУНКЦИИ ===== // Перевод русского текста в руны function russianToRunes(text){ let result = ""; for(let char of text.toUpperCase()){ let found = runes.find(r => r[1] === char); if(found){ result += found[0]; }else{ result += char; } } return result; } // Перевод рун обратно в русский текст function runesToRussian(text){ let result = text; runes.forEach(r => { result = result.split(r[0]).join(r[1]); }); return result; } // Обновлённый переводчик function translateText(){ let text = output.value; if(runeMode){ output.value = runesToRussian(text); modeText.innerHTML = "Режим: Буквы"; }else{ output.value = russianToRunes(text); modeText.innerHTML = "Режим: Руны"; } } // Сохранение избранного let favoriteText = []; function favorites(){ if(output.value.length > 0){ favoriteText.push(output.value); alert("Добавлено в избранное ⭐"); }else{ alert("Нет текста для сохранения"); } } // Просмотр истории let savedHistory = []; function saveHistory(){ if(output.value.length > 0){ savedHistory.push(output.value); } } function historyList(){ if(savedHistory.length === 0){ alert("История пуста"); }else{ alert(savedHistory.join("\n\n")); } } // PNG (простое сохранение через изображение) function downloadPNG(){ let canvas = document.createElement("canvas"); canvas.width = 800; canvas.height = 200; let ctx = canvas.getContext("2d"); ctx.font = "50px Arial"; ctx.fillText(output.value,50,100); let link = document.createElement("a"); link.download = "runopisec.png"; link.href = canvas.toDataURL(); link.click(); } // Настройки function settings(){ let size = prompt( "Введите размер рун (например 38):", "38" ); if(size){ document.querySelectorAll(".rune").forEach(r => { r.style.fontSize = size+"px"; }); } }
